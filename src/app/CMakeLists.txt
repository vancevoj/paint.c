# Lane L2/L4: the paint.c editor.
#
# Every .c file below src/app is compiled into the pc_app library (main.c
# goes into the paintc executable only), so feature lanes extend the app by
# adding files. Two registries are generated from file names:
#   src/app/tools/tool_<name>.c  defines  const app_tool app_tool_<name>;
#   src/app/**/mod_<name>.c      defines  void mod_<name>(app *a);
# (docs/app/ARCHITECTURE.md, "How to add ...").
file(GLOB_RECURSE PC_APP_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.c)
list(REMOVE_ITEM PC_APP_SRCS ${CMAKE_CURRENT_SOURCE_DIR}/main.c)
list(SORT PC_APP_SRCS)

file(GLOB _tools CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/tools/tool_*.c)
list(SORT _tools)
set(_list "/* generated by src/app/CMakeLists.txt - do not edit */\n")
foreach(f ${_tools})
  get_filename_component(_n ${f} NAME_WE)
  string(REGEX REPLACE "^tool_" "" _id ${_n})
  string(APPEND _list "APP_TOOL(${_id})\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gen/app_tool_list.inc CONTENT "${_list}")

file(GLOB_RECURSE _mods CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/mod_*.c)
set(_names)
foreach(f ${_mods})
  get_filename_component(_n ${f} NAME_WE)
  list(APPEND _names ${_n})
endforeach()
list(SORT _names)
set(_list "/* generated by src/app/CMakeLists.txt - do not edit */\n")
foreach(_n ${_names})
  string(REGEX REPLACE "^mod_" "" _id ${_n})
  string(APPEND _list "APP_MOD(${_id})\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gen/app_mod_list.inc CONTENT "${_list}")

# NOTICE embedded for the About dialog (regenerated when NOTICE changes).
include(PcEmbed)
pc_embed_file(PC_APP_EMBED ${PROJECT_SOURCE_DIR}/NOTICE pc_notice_txt)

# ---- lane UIB (wave 4 items 4 to 6): the user guide inside the app ----------
# docs/help/*.md are embedded and rendered to HTML pages at run time
# (src/app/help/help.h), together with the plugin example and the plugin
# headers the Plugins page links to. gen/app_help_list.inc lists the pages.
file(GLOB _help_md CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/docs/help/*.md)
list(SORT _help_md)
set(_list "/* generated by src/app/CMakeLists.txt - do not edit */\n")
foreach(f ${_help_md})
  get_filename_component(_n ${f} NAME_WE)
  string(MAKE_C_IDENTIFIER "${_n}" _id)
  pc_embed_file(PC_APP_EMBED ${f} pc_help_md_${_id})
  string(APPEND _list "APP_HELP_PAGE(${_id}, \"${_n}\")\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gen/app_help_list.inc CONTENT "${_list}")
pc_embed_file(PC_APP_EMBED ${PROJECT_SOURCE_DIR}/docs/help/fade_plugin.c pc_help_fade_plugin_c)
pc_embed_file(PC_APP_EMBED ${PROJECT_SOURCE_DIR}/include/fx/fx_abi.h pc_help_fx_abi_h)
pc_embed_file(PC_APP_EMBED ${PROJECT_SOURCE_DIR}/include/fx/fx_util.h pc_help_fx_util_h)
option(PC_PROJECT_PUBLIC
       "The project repository is public: Help > Forum opens its issue list" OFF)

add_library(pc_app STATIC ${PC_APP_SRCS} ${PC_APP_EMBED})
# lane UIB: Help > Forum exists only for a public repository (mod_help.c)
target_compile_definitions(pc_app PRIVATE PC_PROJECT_PUBLIC=$<BOOL:${PC_PROJECT_PUBLIC}>)
target_include_directories(pc_app PUBLIC ${PROJECT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}
                           PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/gen)
target_link_libraries(pc_app PUBLIC pc_gfx pc_ui pc_pal pc_codec pc_fx pc_core SDL3::SDL3)
pc_warnings(pc_app)
if(NOT MSVC)
  target_link_libraries(pc_app PRIVATE m)
endif()
# lane C: the Text tool's font backend (tools/text_font.c) and IME adapter
# (tools/text_ime.c) use pc_ui internals until ui.h exposes glyph outlines,
# text input and composition; src/ui/ui_font_internal.h includes stb_truetype.h.
target_include_directories(pc_app SYSTEM PRIVATE ${PROJECT_SOURCE_DIR}/third_party/stb_truetype)

# ---- the executable ---------------------------------------------------------
set(PC_APP_PLATFORM_SRCS)
if(WIN32)
  if(MSVC)
    # MSVC and clang-cl merge .manifest sources into the embedded manifest.
    list(APPEND PC_APP_PLATFORM_SRCS ${PROJECT_SOURCE_DIR}/packaging/windows/paintc.manifest)
  else()
    # MinGW: embed the manifest through a resource script.
    enable_language(RC)
    set(PC_APP_MANIFEST ${PROJECT_SOURCE_DIR}/packaging/windows/paintc.manifest)
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/platform/paintc.rc.in
                   ${CMAKE_CURRENT_BINARY_DIR}/paintc.rc @ONLY)
    list(APPEND PC_APP_PLATFORM_SRCS ${CMAKE_CURRENT_BINARY_DIR}/paintc.rc)
  endif()
endif()

add_executable(paintc WIN32 MACOSX_BUNDLE main.c ${PC_APP_PLATFORM_SRCS})
target_link_libraries(paintc PRIVATE pc_app)
pc_warnings(paintc)
if(COMMAND pc_sdl_main)
  pc_sdl_main(paintc)
endif()
if(APPLE)
  configure_file(${CMAKE_CURRENT_SOURCE_DIR}/platform/Info.plist.in
                 ${CMAKE_CURRENT_BINARY_DIR}/Info.plist @ONLY)
  set_target_properties(paintc PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_BINARY_DIR}/Info.plist
    MACOSX_BUNDLE_GUI_IDENTIFIER org.paintc.paintc)
  # ADR-001: the bundle is paint.c.app with the executable paintc inside.
  add_custom_command(TARGET paintc POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E remove_directory ${CMAKE_BINARY_DIR}/paint.c.app
    COMMAND ${CMAKE_COMMAND} -E copy_directory $<TARGET_BUNDLE_DIR:paintc>
            ${CMAKE_BINARY_DIR}/paint.c.app
    VERBATIM)
endif()
