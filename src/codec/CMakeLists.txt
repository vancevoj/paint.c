# Lane L6 (codecs). Every fmt_<id>.c defines `const pc_codec pc_codec_<id>`;
# the registry list is generated from those file names. Other .c files in
# this directory (helpers such as zip.c, quant.c) are compiled in as well.
pc_dep_zlib()
include(PcCodecDeps OPTIONAL)   # lane-owned: libspng, libjpeg-turbo, libwebp, lcms2

pc_glob_sources(PC_CODEC_SRCS ${CMAKE_CURRENT_SOURCE_DIR})
file(GLOB _fmt CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/fmt_*.c)
list(SORT _fmt)
set(_list "/* generated by src/codec/CMakeLists.txt - do not edit */\n")
foreach(f ${_fmt})
  get_filename_component(_n ${f} NAME_WE)
  string(REGEX REPLACE "^fmt_" "" _id ${_n})
  string(APPEND _list "PC_CODEC(${_id})\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gen/pc_codec_list.inc CONTENT "${_list}")

add_library(pc_codec STATIC ${PC_CODEC_SRCS})
target_include_directories(pc_codec PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/gen
                                            ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pc_codec PUBLIC pc_core PRIVATE pc_zlib)
if(COMMAND pc_codec_link_deps)
  pc_codec_link_deps(pc_codec)
endif()
pc_warnings(pc_codec)
