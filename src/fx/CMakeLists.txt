# Lane L5 (effects). Built-in effects compile against include/fx only, like
# plugins. Every file named fxm_<name>.c (any subdirectory) defines
#     int fxm_<name>(const fx_host *host, int (*reg)(const fx_effect *));
# and the built-in module list is generated from those file names.
file(GLOB_RECURSE PC_FX_SRCS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/*.c)
list(SORT PC_FX_SRCS)
file(GLOB_RECURSE _mods CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/fxm_*.c)
list(SORT _mods)
set(_list "/* generated by src/fx/CMakeLists.txt - do not edit */\n")
foreach(f ${_mods})
  get_filename_component(_n ${f} NAME_WE)
  string(APPEND _list "FX_MODULE(${_n})\n")
endforeach()
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gen/fx_module_list.inc CONTENT "${_list}")

add_library(pc_fx STATIC ${PC_FX_SRCS})
target_include_directories(pc_fx PUBLIC ${PROJECT_SOURCE_DIR}/include
                           PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/gen ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(pc_fx PUBLIC pc_core)
pc_warnings(pc_fx)
